یکی از رایج‌ترین اشتباه‌های سرمایه‌گذاران تازه‌کار این است که شرکت و سهم آن را یکی می‌دانند. اما **تفاوت شرکت و سهم** پایه‌ی نگاه بلندمدت به بازار است. شرکت یک کسب‌وکار واقعی است و سهم قیمتی است که بازار هر لحظه برای آن می‌پردازد. در این یادداشت با یک مثال ساده می‌بینید چرا باید «صاحب» را تماشا کنید، نه «سگ» را.

![تشبیه صاحب سگ و قلاده: شرکت آرام حرکت می‌کند، قیمت سهم نوسان می‌کند](/assets/uploads/company-vs-stock-price-1.svg)

## مثال صاحب سگ در پارک

تصور کنید زنی سگ پرانرژی‌اش را در یک مسیر مستقیم در پارک بزرگی قدم می‌زند. از شما می‌خواهند بگویید سگ در هر لحظه کجاست. نکته‌ی سخت این است که قلاده صد پا (حدود سی متر) است. سگ به چپ و راست می‌دود، برمی‌گردد و دوباره می‌رود. پیش‌بینی دقیق موقعیت سگ در دقیقه‌ی بعد تقریباً غیرممکن است.

اما اگر بپرسند «بعد از یک ساعت تقریباً کجا خواهد بود؟»، پاسخ ساده‌تر است: نزدیک جایی که صاحبش به آن‌سوی پارک می‌رسد. چون سگ و صاحب با قلاده به هم وصل‌اند، حرکت کوتاه‌مدت سگ پیش‌بینی‌ناپذیر است ولی مسیر بلندمدت او دنباله‌رو صاحب است. در سرمایه‌گذاری، «کاملاً درست» نبودن کافی است؛ «تقریباً درست» بودن درباره‌ی جهت حرکت بلندمدت، خیلی ارزشمند است.

## شرکت چیست و سهم چیست؟

- **شرکت** گروهی از آدم‌هاست که در دنیای واقعی کالا یا خدمتی ارزشمند تولید می‌کنند. تغییرات آن معمولاً کند است: مشتری جدید، محصول بهتر، هزینه‌ی کمتر، سودآوری بیشتر.
- **سهم** نماینده‌ی نظر دیگران درباره‌ی آینده‌ی شرکت است. این نظر می‌تواند در یک روز، بر اثر خبر، شایعه یا حتی فضای عمومی بازار تغییر کند.

پس شرکت مثل صاحب سگ آرام و پیوسته حرکت می‌کند، ولی سهم مثل سگ در همه‌ی جهت‌ها می‌دود. این دو با قلاده‌ای به هم متصل‌اند: اگر کسب‌وکار در بلندمدت سود بیشتری بسازد، قیمت سهم هم آخرش به آن واکنش نشان می‌دهد.

## قلاده چقدر بلند است؟

طول قلاده در سرمایه‌گذاری همان **قیمتی است که برای ارزش می‌پردازیم**، یعنی ارزشگذاری. اگر برای یک شرکت عالی قیمت بسیار بالایی بپردازید، قلاده خیلی بلند است و سگ می‌تواند خیلی دور از صاحب بدود. در کوتاه‌مدت، همین قیمت پرداخت‌شده مهم‌ترین عامل نتیجه‌ی شماست. در سال‌های پرنوسان بازار، بسیاری از سرمایه‌گذاران دیدند که شرکت خوبی خریده‌اند اما قیمت پرداختی بیش از حد بوده است.

برای شناخت بهتر ابزارهای ارزشگذاری، می‌توانید [اصول مدل مالی خوب](/posts/2022/04/05/good-financial-model-principles/) را بخوانید که چگونگی تکیه بر فرض‌های روشن را توضیح می‌دهد.

## چرا باید «کسب‌وکار» را تماشا کنیم؟

سه دلیل مهم دارد:

1. **سیگنال‌ها در کسب‌وکار روشن‌تر از قیمت‌اند.** رشد فروش، حاشیه‌ی سود، جریان نقدی و رقابت‌پذیری، همه اطلاعات بلندمدت دارند.
2. **قیمت روزانه احساس‌برانگیز است.** تماشای هرروزه‌ی نمودار باعث تصمیم‌های هیجانی می‌شود.
3. **افق بلندمدت به نفع شماست.** اگر مالک شرکتی هستید که مرتب ارزش می‌سازد، نیازی نیست هر ساعت قیمتش را بپرسید.

یک مثال: فرض کنید سهمی ۱۰ درصد در ماه افت کرده است. دو سؤال بپرسید. اول: «آیا فروش، سود یا مزیت رقابتی شرکت تغییر کرده؟» اگر پاسخ منفی است، افت قیمت بیشتر جنبه‌ی نوسان دارد. دوم: «آیا دلیل اولیه‌ی خرید من هنوز برقرار است؟» همین دو پرسش ذهن را از «قیمت» به «کسب‌وکار» می‌برد.

## نکته‌ی عملی: ببینید چه چیزی را تماشا می‌کنید

برای ثبت این عادت، فهرست ساده‌ای بسازید:

- برای هر سرمایه‌گذاری، سه شاخص **کسب‌وکار** بنویسید (مثلاً رشد فروش، حاشیه‌ی سود، بدهی).
- بازبینی آن‌ها را فصلی کنید، نه روزانه.
- قیمت را فقط برای تصمیم خرید یا فروش بررسی کنید، نه برای حال‌وهوا.

این رویکرد با [مغالطه‌ها و سوگیری‌های رایج](/posts/2022/11/06/logical-fallacies-examples/) هم سازگار است؛ چون جلوی تصمیم‌های هیجانی را می‌گیرد.

## پرسش‌های متداول

**آیا قیمت سهم هیچ اهمیتی ندارد؟**
دارد؛ قیمت پرداختی بخشی از بازده را تعیین می‌کند. فقط نباید با نوسان روزانه‌اش شرکت را قضاوت کرد.

**اگر قیمت سال‌ها پایین بماند چه؟**
ممکن است بازار اشتباه کند، اما ممکن است تحلیل شما اشتباه باشد. ببینید کسب‌وکار واقعاً بهتر شده یا نه و فرض‌های خود را بازبینی کنید.

**این رویکرد برای معامله‌گر کوتاه‌مدت هم مناسب است؟**
نه کاملاً؛ این چارچوب مخصوص سرمایه‌گذاری بلندمدت است. معامله‌گر کوتاه‌مدت بیشتر با رفتار قیمت سروکار دارد.

## جمع‌بندی

شرکت و سهم دو چیز متفاوت‌اند که با قلاده‌ای به هم وصل‌اند. قلاده را «ارزشگذاری» تعیین می‌کند و جهت بلندمدت را «کسب‌وکار». سرمایه‌گذار بلندمدت صاحب را تماشا می‌کند. این هفته برای یکی از دارایی‌های خود سه شاخص کسب‌وکار بنویسید و فقط همان‌ها را دنبال کنید.
